Demo 02 · CounterState (useState & functional updates)

The fixed counter with useState. The last two buttons call the setter three times: once with the current value and once with the functional form. 

function CounterState() {
  const [count, setCount] = useState(0);

  const increment = () => setCount((prev) => prev + 1);
  const decrement = () => setCount((prev) => prev - 1);
  const reset = () => setCount(0);

  function plusThreeStale() {
    setCount(count + 1);
    setCount(count + 1);
    setCount(count + 1);   // all three use the same snapshot
  }

  function plusThreeFunctional() {
    setCount((prev) => prev + 1);
    setCount((prev) => prev + 1);
    setCount((prev) => prev + 1);   // each one gets the latest value
  }

  return (
    <div>
      <p aria-live="polite">Count: <strong>{count}</strong></p>
      <button onClick={decrement}>−</button>
      <button onClick={increment}>+</button>
      <button onClick={reset}>Reset</button>
      <button onClick={plusThreeStale}>3 × setCount(count + 1)</button>
      <button onClick={plusThreeFunctional}>3 × setCount(prev => prev + 1)</button>
    </div>
  );
}
