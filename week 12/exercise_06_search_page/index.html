Demo 06 · SearchPage (lifting state up)

SearchPage owns query and results. SearchBar sends each change up through onSearch; ResultList only reads props. 

const BERRIES = ['Blackberry', 'Blueberry', 'Cloudberry',
  'Cranberry', 'Elderberry', 'Gooseberry', 'Huckleberry',
  'Mulberry', 'Raspberry', 'Salmonberry', 'Strawberry'];

const fetchResults = (q) =>
  Promise.resolve(BERRIES.filter((b) =>
    b.toLowerCase().includes(q.trim().toLowerCase())
  )));

function SearchPage() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState(BERRIES);

  const handleSearch = async (q) => {
    setQuery(q);
    const data = await fetchResults(q);
    setResults(data);
  };

  return (
    <>
      <SearchBar onSearch={handleSearch} />
      <ResultList query={query} results={results} />
    </>
  );
}
