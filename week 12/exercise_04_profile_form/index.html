Demo 04 · ProfileForm (state with an object)

One object in state, updated with the spread operator so every change creates a new object. 

function ProfileForm() {
  const [user, setUser] = useState({ name: '', email: '', bio: '' });

  const handleChange = (field, value) => {
    setUser((prev) => ({ ...prev, [field]: value }));
  };

  return (
    <form onSubmit={(e) => e.preventDefault()}>
      <label htmlFor="pf-name">Name</label>
      <input id="pf-name" value={user.name}
             onChange={(e) => handleChange('name', e.target.value)} />

      <label htmlFor="pf-email">Email</label>
      <input id="pf-email" type="email" value={user.email}
             onChange={(e) => handleChange('email', e.target.value)} />

      <label htmlFor="pf-bio">Bio</label>
      <textarea id="pf-bio" rows="3" value={user.bio}
                onChange={(e) => handleChange('bio', e.target.value)} />

      <p>State object:</p>
      <pre><code>{JSON.stringify(user, null, 2)}</code></pre>
    </form>
  );
}
