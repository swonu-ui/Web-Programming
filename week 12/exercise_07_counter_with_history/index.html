Demo 07 · Counter with History

Add 1, subtract 1, reset, view the last five values, and undo. This demo uses two state variables: count and history. 

function CounterWithHistory() {
  const [count, setCount] = useState(0);
  const [history, setHistory] = useState([0]);

  function change(next) {
    setCount(next);
    setHistory((prev) => [...prev, next]);
  }

  function undo() {
    if (history.length < 2) return;
    setCount(history[history.length - 2]);
    setHistory((prev) => prev.slice(0, -1));
  }

  return (
    <div>
      <p>{count}</p>
      <button onClick={() => change(count + 1)}>+1</button>
      <button onClick={() => change(count - 1)}>−1</button>
      <button onClick={() => change(0)}>Reset</button>
      <button onClick={undo} disabled={history.length < 2}>Undo</button>
      <p>History (last 5): [{history.slice(-5).join(', ')}]</p>
    </div>
  );
}
