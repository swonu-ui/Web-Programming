Demo 03 · UserForm (multiple useState calls)

Four separate state variables: name, email, age and submitted. 

function UserForm() {
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [age, setAge] = useState(0);
  const [submitted, setSubmitted] = useState(false);

  function handleSubmit(e) {
    e.preventDefault();
    setSubmitted(true);
  }

  if (submitted) {
    return (
      <div>
        <p>Thanks, {name || 'friend'}! Email: {email || '(none)'}, Age: {age}</p>
        <button onClick={() => setSubmitted(false)}>Edit again</button>
      </div>
    );
  }

  return (
    <form onSubmit={handleSubmit}>
      <label htmlFor="uf-name">Name</label>
      <input id="uf-name" value={name} onChange={e => setName(e.target.value)} />
      <label htmlFor="uf-email">Email</label>
      <input id="uf-email" type="email" value={email} onChange={e => setEmail(e.target.value)} />
      <label htmlFor="uf-age">Age</label>
      <input id="uf-age" type="number" value={age} onChange={e => setAge(Number(e.target.value))} />
      <button type="submit">Submit</button>
    </form>
  );
}
