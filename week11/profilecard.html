Build a ProfileCard component

// Default avatar: an inline SVG circle, so the card works offline too
const DEFAULT_AVATAR = 'data:image/svg+xml,' + encodeURIComponent(
  '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 80 80"><circle cx="40" cy="40" r="40" fill="#0e7490"/></svg>'
);

function ProfileCard({ name, role, email, avatarUrl = DEFAULT_AVATAR }) {
  return (
    <div className="profile-card">
      <img src={avatarUrl} alt="" width="80" height="80" />
      <h2>{name}</h2>
      <p>{role}</p>
      <a href={`mailto:${email}`}>{email}</a>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(
  <ProfileCard
    name="Dr. Toshpulatov"
    role="Professor"
    email="muhiddin@gachon.ac.kr"
  />
);

The avatar has alt="" because the name is already written below it, so the image is decorative.
