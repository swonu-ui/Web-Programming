Layout with props.children

function Card({ title, children }) {
  return (
    <section className="content-card">
      <h3>{title}</h3>
      {children}
    </section>
  );
}

function App() {
  return (
    <div className="card-stack">
      <Card title="Lecture Topics">
        <ul>
          <li>Components</li>
          <li>Props</li>
        </ul>
      </Card>

      <Card title="Homework">
        <p>
          Build a student profile card component
        </p>
      </Card>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(
  <App />
);

The children prop lets Card wrap any content inside it.
